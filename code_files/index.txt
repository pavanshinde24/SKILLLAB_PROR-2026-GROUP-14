<script>
    // Connect to Python Flask Backend using Socket.IO
    const socket = io();
    let currentState = 'LOCKED';
    
    // Telemetry Simulation Data
    let odo = 1452.4;
    let batt = 87;
    
    setInterval(() => {
        if(currentState === 'UNLOCKED') {
            odo += 0.1; batt = Math.max(0, batt - 0.5);
            document.getElementById('odo-val').innerText = odo.toFixed(1);
            document.getElementById('battery-val').innerText = Math.floor(batt) + '%';
            document.getElementById('battery-bar').style.width = batt + '%';
        }
    }, 3000);

    // Listen for real-time hardware updates from Python
    socket.on('car_update', function(data) {
        currentState = data.state;
        updateUI();
        if(data.message && data.state === 'ALARM') alert("Warning: " + data.message);
    });

    socket.on('show_contact', function() {
        document.getElementById('contact-panel').classList.remove('hidden');
    });

    // Send Commands to Python
    function sendCommand(action) {
        fetch('/api/command', {
            method: 'POST',
            headers: {'Content-Type': 'application/json'},
            body: JSON.stringify({action: action})
        });
    }

    // OTP Logic
    function toggleLockState() {
        if(currentState === 'LOCKED') {
            document.getElementById('normal-controls').classList.add('hidden');
            document.getElementById('otp-form').classList.remove('hidden');
        } else {
            sendCommand('lock');
        }
    }

    function cancelOtp() {
        document.getElementById('normal-controls').classList.remove('hidden');
        document.getElementById('otp-form').classList.add('hidden');
        document.getElementById('otp-input').value = '';
    }

    function submitOtp() {
        const otp = document.getElementById('otp-input').value;
        fetch('/api/otp', {
            method: 'POST',
            headers: {'Content-Type': 'application/json'},
            body: JSON.stringify({otp: otp})
        }).then(res => res.json()).then(data => {
            if(data.status === 'success') {
                cancelOtp();
            } else {
                alert("Invalid OTP!");
            }
        });
    }

    // Simulator for Hackathon Presentation
    function simulateEvent(event) {
        fetch('/api/simulate', {
            method: 'POST',
            headers: {'Content-Type': 'application/json'},
            body: JSON.stringify({event: event})
        });
    }

    // UI Updater
    function updateUI() {
        const banner = document.getElementById('alert-banner');
        const icon = document.getElementById('alert-icon');
        const title = document.getElementById('alert-title');
        const desc = document.getElementById('alert-desc');
        const dot = document.getElementById('status-dot');
        const text = document.getElementById('status-text');
        const mainBtn = document.getElementById('main-btn');

        banner.classList.remove('hidden', 'bg-red-600', 'bg-green-500/20', 'bg-slate-800', 'animate-pulse');
        icon.className = '';

        text.innerText = currentState;

        if(currentState === 'ALARM') {
            banner.classList.add('bg-red-600', 'animate-pulse');
            icon.classList.add('fas', 'fa-shield-virus', 'text-white', 'text-3xl');
            title.innerHTML = '<span class="text-white">SECURITY ALERT</span>';
            desc.innerHTML = '<span class="text-red-100">Tampering detected! Immobilizer active.</span>';
            dot.className = 'w-3 h-3 rounded-full bg-red-500 animate-ping';
            mainBtn.innerHTML = '<i class="fas fa-unlock"></i> Emergency OTP Unlock';
            mainBtn.className = 'w-full bg-blue-600 hover:bg-blue-700 py-4 rounded-xl font-bold flex items-center justify-center gap-2 transition-all text-white';
        } else if (currentState === 'UNLOCKED') {
            banner.classList.add('bg-green-500/20', 'border', 'border-green-500/50');
            icon.classList.add('fas', 'fa-unlock-alt', 'text-green-400', 'text-3xl');
            title.innerHTML = '<span class="text-green-400">SYSTEM ONLINE</span>';
            desc.innerHTML = '<span class="text-green-200">Car unlocked. Motors enabled.</span>';
            dot.className = 'w-3 h-3 rounded-full bg-green-500';
            mainBtn.innerHTML = '<i class="fas fa-lock"></i> Remote Lock Vehicle';
            mainBtn.className = 'w-full bg-slate-800 hover:bg-slate-700 py-4 rounded-xl font-bold flex items-center justify-center gap-2 transition-all text-white';
        } else {
            banner.classList.add('bg-slate-800', 'border', 'border-slate-700');
            icon.classList.add('fas', 'fa-lock', 'text-slate-400', 'text-3xl');
            title.innerHTML = '<span class="text-slate-200">SECURED</span>';
            desc.innerHTML = '<span class="text-slate-400">Car locked. Alarm armed.</span>';
            dot.className = 'w-3 h-3 rounded-full bg-red-500';
            mainBtn.innerHTML = '<i class="fas fa-unlock"></i> Emergency OTP Unlock';
            mainBtn.className = 'w-full bg-blue-600 hover:bg-blue-700 py-4 rounded-xl font-bold flex items-center justify-center gap-2 transition-all text-white';
        }
    }
    
    // Initial Setup
    updateUI();
</script>
